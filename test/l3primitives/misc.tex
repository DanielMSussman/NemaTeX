% Test file for simple l3kernel utility primitives:
%   \pageheight, \pagewidth     
%   \resettimer, \elapsedtime  
%   \shellescape                 
%   \setrandomseed, \randomseed 
%   \savepos, \lastxpos, \lastypos
%% Public domain.
%% Daniel M Sussman (2026)

\catcode`\{=1 \catcode`\}=2 \catcode`\#=6
\def\typ#1{\immediate\write-1 {#1}}

\typ{START}

\typ{TEST 1: pagewidth and pageheight}
\pagewidth=500pt
\pageheight=700pt
\typ{\the\pagewidth}
\typ{\the\pageheight}

\typ{TEST 2: resettimer and elapsedtime}
\resettimer
\ifnum\elapsedtime<0
  \typ{FAIL: elapsed time negative}
\else
  \typ{PASS: elapsed time non-negative}
\fi

\typ{TEST 3: shellescape}
\typ{\the\shellescape}
\ifnum\shellescape=0
  \typ{PASS: shellescape is 0}
\else
  \typ{FAIL: shellescape is not 0}
\fi

\typ{TEST 4: setrandomseed and randomseed}
\setrandomseed 12345
\typ{\the\randomseed}
\ifnum\randomseed=12345
  \typ{PASS: random seed matches}
\else
  \typ{FAIL: random seed mismatch}
\fi

\typ{TEST 5: Position tracking}
\ifnum\lastxpos=0
  \typ{PASS: initial lastxpos is 0}
\else
  \typ{FAIL: initial lastxpos non-zero}
\fi
\ifnum\lastypos=0
  \typ{PASS: initial lastypos is 0}
\else
  \typ{FAIL: initial lastypos non-zero}
\fi

\shipout\hbox{\savepos}
\ifnum\lastxpos>0
  \typ{PASS: lastxpos is positive after shipout}
\else
  \typ{FAIL: lastxpos non-positive}
\fi
\ifnum\lastypos>0
  \typ{PASS: lastypos is positive after shipout}
\else
  \typ{FAIL: lastypos non-positive}
\fi

\count10=\lastxpos
\shipout\hbox{\kern 100pt\savepos}
\count11=\lastxpos
\advance\count11 by -\count10
\ifnum\count11=6553600
  \typ{PASS: exact 100pt horizontal displacement}
\else
  \typ{FAIL: unexpected horizontal displacement}
\fi

\pdfhorigin=2in
\shipout\hbox{\savepos}
\ifnum\lastxpos=\pdfhorigin
  \typ{PASS: custom pdfhorigin respected}
\else
  \typ{FAIL: custom pdfhorigin not respected}
\fi

\pdfvorigin=1in
\pageheight=10in
\shipout\vbox{\kern 1in\savepos}
\count10=\pageheight
\advance\count10 by -\pdfvorigin
\advance\count10 by -\dimexpr 1in\relax
\ifnum\lastypos=\count10
  \typ{PASS: custom pageheight and pdfvorigin respected}
\else
  \typ{FAIL: custom pageheight and pdfvorigin not respected}
\fi

\typ{END}
\end
