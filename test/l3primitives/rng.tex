% Test file for RNG primitives (assumes the pdftex prng implementation):
%   \setrandomseed  -- set the random seed
%   \randomseed     -- read current seed
%   \uniformdeviate -- uniform random integer in [0, N)
%   \normaldeviate  -- Gaussian-distributed random integer
%% Public domain.
%% Daniel M Sussman (2026)

\catcode`\{=1 \catcode`\}=2 \catcode`\#=6
\def\typ#1{\immediate\write-1 {#1}}

\typ{START}

\typ{RNG TEST 1: setrandomseed and randomseed}
% Setting a seed and reading it back
\setrandomseed 42
\typ{\the\randomseed}

\typ{RNG TEST 2: uniform deviate with seed}
% With a fixed seed, the sequence must be deterministic
\setrandomseed 12345
\typ{\uniformdeviate 100}
\typ{\uniformdeviate 100}
\typ{\uniformdeviate 100}

\typ{RNG TEST 3: uniform deviate range}
% \uniformdeviate N returns value in [0, N)
% With N=1, must always be 0
\setrandomseed 99999
\typ{\uniformdeviate 1}
\typ{\uniformdeviate 1}

\typ{RNG TEST 4: normal deviate with seed}
% With a fixed seed, the sequence must be deterministic
\setrandomseed 12345
\typ{\normaldeviate}
\typ{\normaldeviate}
\typ{\normaldeviate}

\typ{RNG TEST 5: reseed changes sequence}
% Different seeds give different sequences
\setrandomseed 1
\edef\seqA{\uniformdeviate 1000000}
\setrandomseed 2
\edef\seqB{\uniformdeviate 1000000}
\ifnum\seqA=\seqB
  \typ{FAIL: different seeds gave same value}
\else
  \typ{PASS: different seeds gave different values}
\fi

\typ{END}
\end
